\begin{align*}
	c_i&=\sum_{j=1}w_j\cdot h_j\\
	h_1^{\text{next}}&=\sum_{i=1}w_i\cdot c_i\\
	e_1^{\text{next}}&=a_1^\text{next}-h_1^\text{next}\\
	E&=\frac{\sum_{k=1}(e_k^{\text{next}})^2}{2}\\
	\frac{dE}{dw_j}&=\frac{dE}{de_k}\cdot \frac{de_k}{dh_k} \cdot \frac{dh_k}{dc_i} \cdot \frac{dc_i}{dw_j}\\
\end{align*}

\begin{align*}
	\frac{dc_i}{dw_j}&=\frac{((w_j+d)\cdot h_j+r)-(w_j\cdot h_j+r)}{d}\\
	\frac{dc_i}{dw_j}&=\frac{(w_jh_j+dh_j+r)-(w_jh_j+r)}{d}\\
	\frac{dc_i}{dw_j}&=\frac{\cancel{w_jh_j}+dh_j+\cancel{r}-\cancel{w_jh_j}-\cancel{r}}{d}\\
	\frac{dc_i}{dw_j}&=\frac{\cancel{d}h_j}{\cancel{d}}\\
	\frac{dc_i}{dw_j}&=h_j\\
\end{align*}

\begin{align*}
	\frac{dh_k}{dc_i}&=\frac{((w_i+d)\cdot c_i+r)-(w_i\cdot c_i+r)}{d}\\
	\frac{dh_k}{dc_i}&=\frac{(w_ic_i+dc_i+r)-(w_ic_i+r)}{d}\\
	\frac{dh_k}{dc_i}&=\frac{\cancel{w_ic_i}+dc_i+\cancel{r}-\cancel{w_ic_i}-\cancel{r}}{d}\\
	\frac{dh_k}{dc_i}&=\frac{\cancel{d}c_i}{\cancel{d}}\\
	\frac{dh_k}{dc_i}&=c_i\\
\end{align*}

\begin{align*}
	\frac{de_k}{dh_k}&=\frac{(a_k^\text{next}- (h_k^\text{next}+d))-(a_k^\text{next}- h_k^\text{next})}{d}\\
	\frac{de_k}{dh_k}&=\frac{a_k^\text{next}- h_k^\text{next}-d-a_k^\text{next}+ h_k^\text{next}}{d}\\
	\frac{de_k}{dh_k}&=\frac{\cancel{a_k^\text{next}}- \cancel{h_k^\text{next}}-d-\cancel{a_k^\text{next}}+ \cancel{h_k^\text{next}}}{d}\\
	\frac{de_k}{dh_k}&=\frac{-\cancel{d}}{\cancel{d}}\\
	\frac{de_k}{de_k}&=-1
\end{align*}

\begin{align*}
	\frac{dE}{de_k}&=\frac{((e_k^\text{next}+d)^2+r)-((e_k^\text{next})^2+r)}{2d}\\
	\frac{dE}{de_k}&=\frac{((e_k^\text{next})^2+2e_k^\text{next}d+d^2+r)-((e_k^\text{next})^2+r)}{2d}\\
	\frac{dE}{de_k}&=\frac{(e_k^\text{next})^2+2e_k^\text{next}d+d^2+r-(e_k^\text{next})^2-r}{2d}\\
	\frac{dE}{de_k}&=\frac{\cancel{(e_k^\text{next})^2}+2e_k^\text{next}d+d^2+\cancel{r}-\cancel{(e_k^\text{next})^2}-\cancel{r}}{2d}\\
	\frac{dE}{de_k}&=\frac{2e_k^\text{next}\cancel{d}+\cancel{d}^2}{2\cancel{d}}\\
	\frac{dE}{de_k}&=\frac{2e_k^\text{next}+d}{2}\\
	\lim_{d\to 0}&=\frac{\cancel{2}e_k^\text{next}}{\cancel{2}}\\
	\frac{dE}{de_k}&=e_k^\text{next}\\
\end{align*} 

\begin{align*}

\begin{align*}
\diff{d}{w^{h_6}} &= \diff{d}{o} \cdot \diff{o}{h_6} \cdot \diff{h_6}{w^{h_6}}\\
\diff{d}{w^{h_6}} &= -1\cdot w_3^o \cdot 1=-w_3^o\\
\Delta w^{h_6} &= -\diff{d}{w^{h_6}}=-(-w_3^o)=w_3^o\\
h_6&=\underbrace{h_1\cdot w_1^{h_6}}_{r}+ 1\cdot w_2^{h_6}\\
\diff{h_6}{w}&=\frac{(r+1\cdot (w_2^{h_6}+d))-(r+1\cdot w_2^{h_6})}{d}\\
&=\frac{(r+w_2^{h_6}+d)-(r+1\cdot w_2^{h_6})}{d}\\
&=\frac{\cancel{r}+\cancel{w_2^{h_6}}+d-\cancel{r}-\cancel{w_2^{h_6}}}{d}\\
&=\frac{d}{d}\\
&=1
\end{align*}

\begin{align*}
o&=\underbrace{h_7\cdot w_1^o+1\cdot w_2^o}_{r}+ h_6\cdot w_3^o\\
\diff{o}{h_6}&=\frac{(r+(h_6+d)\cdot w_3^o)-(r+h_6\cdot w_3^o)}{d}\\
&=\frac{(r+h_6w_3^o+dw_3^o)-(r+h_6\cdot w_3^o)}{d}\\
&=\frac{r+h_6w_3^o+dw_3^o-r-h_6w_3^o}{d}\\
&=\frac{\cancel{r}+\cancel{h_6w_3^o}+dw_3^o-\cancel{r}-\cancel{h_6w_3^o}}{d}\\
&=\frac{\cancel{d}w_3^o}{\cancel{d}}\\
&=w_3^o
\end{align*}

\begin{align*}
	d&=\frac{(p-o)^2}{2}\\
	\diff{d}{o}&=\frac{(p-(o+d))^2-(p-o)^2}{2d}\\
	&=\frac{(p-o-d)^2-(p-o)^2}{2d}\\
	&=\frac{(p^2-2po-2pd+o^2+2od+d^2)-(p^2-2po+o^2)}{2d}\\
	&=\frac{\cancel{p^2}-\cancel{2po}-2pd+\cancel{o^2}+2od+d^2-\cancel{p^2}+\cancel{2po}-\cancel{o^2}}{2d}\\
	&=\frac{-2p\cancel{d}+2o\cancel{d}+\cancel{d}^2}{2\cancel{d}}\\
	&=\frac{-\cancel{2}p+\cancel{2}o+d}{\cancel{2}}\\
	\lim_{d\to 0}&=-p+o+d=-p+o
\end{align*}

\begin{align*}
	d&=p-o\\
	\diff{d}{o}&=\frac{(p-(o+d))-(p-o)}{d}\\
	&=\frac{p-o-d-p+o}{d}\\
	&=\frac{\cancel{p}-\cancel{o}-d-\cancel{p}+\cancel{o}}{d}\\
	&=\frac{-d}{d}\\
	&=-1
\end{align*}
